#!/usr/bin/env python3
"""Lab 2 — turning a model into a software component.

Two ways to get structured data out of an API model, compared head to head:

  A. Ask nicely in the prompt, then parse.       (fragile)
  B. Define a schema and force a tool call.      (reliable)

    python structured.py                 # both, 3 trials each
    python structured.py --tier 2        # only the ambiguous tickets
    python structured.py --trials 10     # sharper contrast
    python structured.py --misses        # show which trap caught it

Tickets, schema and scoring live in tickets.py, shared with
structured_local.py so the API and local numbers are directly comparable.
"""
 - Shebang: enables direct execution from terminal. - Docstring explains the experiment: comparing two extraction methods across randomized trials. - Lists CLI usage variants; documents that shared test cases live in tickets.py for consistency with a local-run companion script. 
import argparse
import json
import os

import anthropic
from dotenv import load_dotenv

import tickets as T
 - argparse: parse CLI flags. - json: serialize/deserialize model output. - os: read environment variables (model name, keys). - anthropic: official Anthropic SDK for Claude. - dotenv: load .env with secrets; keeps them out of code. - tickets as T: shared module with test cases, schema, validation helpers, and prompt templates. 
load_dotenv()

MODEL = os.environ.get("WORKSHOP_MODEL", "claude-sonnet-5")
client = anthropic.Anthropic()
 - load_dotenv: loads secrets (API key) from .env into environment. - MODEL: configurable model choice; defaults to claude-sonnet-5 for workshop consistency. - Create Anthropic client once, reused by both methods. 
TOOL = {
    "name": "record_ticket",
    "description": "Record a parsed support ticket in the tracking system.",
    "input_schema": T.SCHEMA,
}
 - Defines the tool used by Method B. - input_schema (from tickets.py) specifies required fields and types for each ticket record; Claude’s output must match this shape or the API rejects it as invalid arguments. 
# --------------------------------------------------------------------------
# Method A — ask in the prompt
# --------------------------------------------------------------------------
 - Marker comment; introduces “prompt-only” approach. 
def method_a(text: str):
    resp = client.messages.create(
        model=MODEL, max_tokens=600,
        messages=[{"role": "user",
                   "content": T.PROMPT_A.format(rules=T.RULES, ticket=text)}],
    )
 - Sends the raw ticket text plus instructions (PROMPT_A) that ask Claude to “format as JSON.” - No structural guarantees—just natural language constraints. 
    out = "".join(b.text for b in resp.content if b.type == "text").strip()

    # The defensive cleanup every codebase ends up with. This is the smell.
    if out.startswith("```"):
        out = out.split("```")[1].removeprefix("json").strip()
 - Concatenates text blocks and trims whitespace. - Then defensively strips markdown fences (```` ``` ... ````) and possible “json” tag—classic brittle cleanup that appears whenever you rely on prompt-only formatting. 
    try:
        return json.loads(out)
    except json.JSONDecodeError:
        return None
 - Attempts JSON parse of cleaned output. - On failure, returns None instead of crashing—so failures become measurable “invalid” results rather than hard errors. 
# --------------------------------------------------------------------------
# Method B — declare a tool, force the call
# --------------------------------------------------------------------------
 - Marker comment for schema-driven approach. 
def method_b(text: str):
    resp = client.messages.create(
        model=MODEL, max_tokens=600, tools=[TOOL],
        # The whole trick: the model MUST call this tool, and the API validates
        # its arguments against the schema before you ever see them.
        tool_choice={"type": "tool", "name": "record_ticket"},
        messages=[{"role": "user",
                   "content": f"{T.RULES}\n\nParse this support ticket:\n\n{text}"}],
    )
 - Supplies the TOOL definition and forces its use via tool_choice. - Because of tool_choice, Claude is required to produce a call(record_ticket) with arguments conforming to SCHEMA; otherwise the API rejects it at transport time. - The prompt itself can be minimal since structure enforcement happens at the schema layer. 
    for block in resp.content:
        if block.type == "tool_use":
            return block.input
    return None
 - Iterates response blocks; when a tool_use block appears, its input is already validated against the schema and returned as a structured dict. - If no tool_use occurs, returns None (treated as invalid attempt). 
METHODS = [("A: prompt-based", method_a), ("B: tool schema", method_b)]
 - Centralized list of (label, callable) so the runner can treat both approaches uniformly. 
def run(trials: int, tier: str, misses: bool) -> None:
    cases = T.select(tier)
    tiers = sorted({c["tier"] for c in cases})
    print(f"\n  model {MODEL}   tickets {len(cases)} (tier {tier})   trials {trials}\n")
 - run: core experiment harness. - T.select(tier): loads test tickets by difficulty tier (“1”, “2”, or all). - Prints header with model, ticket count, and trial budget so logs are self-contained. 
    header = f"  {'method':<18} {'valid':>13} {'fields right':>14}"
    for ti in tiers:
        header += f" {'tier ' + str(ti):>9}"
    print(header)
    print("  " + "-" * (len(header) - 2))
 - Constructs a columnar header: - “valid”: how many attempts yielded parseable/valid output. - “fields right”: accuracy of individual fields vs ground truth. - Per-tier columns to see where each method struggles most. 
    all_misses = []
    for name, fn in METHODS:
        valid = attempts = got = possible = 0
        per_tier = {ti: [0, 0] for ti in tiers}
 - Initializes counters and per-tier tracking arrays: - valid/ attempts → structural validity rate. - got/possible → correct field count vs total fields attempted. - all_misses collects specifics of failures when --misses is requested. 
        for case in cases:
            for _ in range(trials):
                attempts += 1
                try:
                    result = fn(case["text"])
                except Exception as exc:  # noqa: BLE001
                    all_misses.append((name, case["id"], type(exc).__name__))
                    continue
 - For each ticket and for each trial: - Calls the method (fn) on that ticket text. - On unexpected exception (network blip, etc.), logs it and continues instead of aborting—keeps experiment robust to transient errors. 
                if T.is_valid(result):
                    valid += 1
                s, total, wrong = T.accuracy(result, case["expect"])
                got += s
                possible += total
                per_tier[case["tier"]][0] += s
                per_tier[case["tier"]][1] += total
                for w in wrong:
                    all_misses.append((name, case["id"], w))
 - T.is_valid(result): checks whether the returned structure is acceptable (not None, has required fields). - T.accuracy(...): compares result to ground-truth expect, returning: - s = score (correct field count) - total = fields evaluated - wrong = list of failure reasons/traps. - Tallies both overall and per-tier; tracks each specific miss for later diagnostics when --misses is used. 
        vpct = 100 * valid / attempts if attempts else 0
        apct = 100 * got / possible if possible else 0
        row = f"  {name:<18} {valid}/{attempts} ({vpct:>3.0f}%)".ljust(34)
        row += f"{got}/{possible} ({apct:>3.0f}%)".rjust(14)
        for ti in tiers:
            g, p = per_tier[ti]
            row += f" {(100 * g / p if p else 0):>8.0f}%"
        print(row)
 - Computes and prints one summary row per method: - Validity rate (vpct) shows structural reliability. - Field accuracy (apct) shows correctness of content. - Per-tier columns highlight how difficulty affects each approach. - This is where you see Method B’s advantage clearly in practice. 
    if misses and all_misses:
        print("\n  \033[93mwhat went wrong\033[0m")
        seen = {}
        for method, cid, what in all_misses:
            seen.setdefault((method, cid), []).append(what)
        for (method, cid), whats in sorted(seen.items()):
            case = next(c for c in cases if c["id"] == cid)
            print(f"    {method:<18} {cid:<20} {max(set(whats), key=whats.count)}")
            print(f"    {'':<18} \033[90mtrap: {case['traps']}\033[0m")
 - If --misses is set, prints a compact diagnostic of recurring issues. - Groups by (method, case), picks the most common failure mode for that pair, and annotates which “trap” in the ticket caused it—useful when teaching or tuning prompts. 
    print("""
  Method B cannot produce a malformed record — the API validates arguments
  against the schema first. It can still produce a WRONG one. Compare the
  tier 2 column against tier 1 to see the difference, then run
  structured_local.py and compare against your own GPU.
""")
 - Prints key takeaway: - Method B guarantees well-formed output (schema-enforced), though not guaranteed correctness of content on hard cases. - Encourages comparing tiers to see where each method degrades, and invites running structured_local.py for local vs cloud model comparisons. 
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--trials", type=int, default=3)
    ap.add_argument("--tier", default="all", choices=["1", "2", "all"])
    ap.add_argument("--misses", action="store_true")
    args = ap.parse_args()
    run(args.trials, args.tier, args.misses)
 - CLI entrypoint wiring: - --trials N: number of independent attempts per ticket (more trials → sharper stats). - --tier {1,2,all}: filter test cases by difficulty. - --misses: enable detailed failure diagnostics. - Delegates to run with parsed arguments; keeps experiment logic separate from wiring.
