def tool_list_files(pattern: str = "*") -> str:
    files = sorted(p for p in WORKSPACE.glob(pattern) if p.is_file())
    if not files:
        return f"No files matching {pattern!r}."
    return "\n".join(f"{p.relative_to(WORKSPACE)} ({p.stat().st_size} bytes)" for p in files)
 Implements list_files: glob WORKSPACE, filter to files only, and return a sorted, human-readable summary with sizes. 
def tool_read_file(path: str) -> str:
    target = (WORKSPACE / path).resolve()

    # Containment check. Without this, path='../../.env' reads your API key.
    # Note where this lives: in YOUR code, not in the model's instructions.
    if not target.is_relative_to(WORKSPACE.resolve()):
        return "ERROR: path escapes the workspace. Refused."
    if not target.exists():
        return f"ERROR: {path} does not exist."

    content = target.read_text(errors="replace")
    if len(content) > 8000:
        return content[:8000] + f"\n\n[truncated — file is {len(content)} chars]"
    return content
 Implements read_file with safety: resolves absolute path, ensures it never escapes WORKSPACE (security guard), checks existence, reads text safely, and truncates very large files to avoid token overflow. 
def tool_calculate(expression: str) -> str:
    allowed = set("0123456789+-*/(). ")
    if not set(expression) <= allowed:
        return "ERROR: expression contains disallowed characters."
    try:
        return str(eval(expression, {"__builtins__": {}}, {}))  # noqa: S307
    except Exception as exc:  # noqa: BLE001
        return f"ERROR: {exc}"
 Implements calculate with a strict allowlist and empty builtins to run only numeric expressions; catches errors instead of leaking stack traces.
